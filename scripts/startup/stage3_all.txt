# Стартовый скрипт этапа 3: проверка всех команд этапов 1-3.
# Сообщение из файла motd (если оно есть в корне VFS) показывается
# при запуске, до выполнения этого скрипта.

# --- служебная команда vfs-info: сведения о загруженной VFS ---
vfs-info
vfs-info extra     # ошибка: команда не принимает аргументов

# --- команды ls и cd работают с загруженной VFS ---
ls
ls -l /
ls -a /
cd /
cd
cd ..   # комментарий в конце строки

# --- обработка ошибок ---
foo
unknown_command 1 2 3

# --- пустые строки и комментарии пропускаются ---

# --- завершение работы ---
exit
ls   # эта команда не выполнится: после exit скрипт останавливается
