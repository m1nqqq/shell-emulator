# Стартовый скрипт этапа 2.
# Строки, начинающиеся с символа #, считаются комментариями.
# Остальные строки выполняются, а ввод и вывод показываются на экране.

ls
ls -l /tmp
cd docs   # комментарий в конце строки тоже игнорируется
cd

# пустые строки и комментарии не выполняются
ls   a    b
exit
echo эта строка не выполняется: после exit скрипт останавливается
