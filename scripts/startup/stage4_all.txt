# Стартовый скрипт этапа 4: команды ls, cd, cal, whoami и find.
# Запуск: python src/main.py --vfs scripts/vfs/deep.csv --script scripts/startup/stage4_all.txt
# Скрипт рассчитан на VFS scripts/vfs/deep.csv (шесть уровней вложенности).

# ===== whoami =====
whoami
whoami now      # ошибка: лишний аргумент

# ===== ls: все режимы =====
ls              # текущий каталог
ls -a           # со скрытыми файлами
ls -l           # подробный формат
ls /etc /var    # несколько каталогов
ls /etc/config.cfg   # файл вместо каталога
ls /tmp         # пустой каталог
ls -la /home/user    # объединённые опции и скрытый .bashrc
ls nowhere      # ошибка: нет такого пути
ls -z           # ошибка: неизвестная опция

# ===== cd: все режимы =====
cd home/user                  # относительный путь
cd projects/emulator/src      # вложенный каталог, приглашение меняется
ls -l
cd ..                         # на уровень вверх
cd ../../docs                 # несколько уровней вверх
ls notes
cd /var/log                   # абсолютный путь
cd ~                          # домашний каталог (корень VFS)
cd /home/user/docs
cd                            # без аргумента: в корень
cd ../../..                   # выше корня не подняться
cd /etc/config.cfg            # ошибка: это файл
cd /nowhere                   # ошибка: нет такого каталога
cd a b                        # ошибка: слишком много аргументов

# ===== find: все режимы =====
find                                    # от текущего каталога
find /home/user/projects                # от указанного пути
find /home -name "*.txt"                # по шаблону имени
find / -type d                          # только каталоги
find /home/user -type f -name "*.py"    # тип и имя вместе
find /etc /var -type f                  # несколько путей
find nowhere                            # ошибка: нет такого пути
find -name                              # ошибка: нет значения
find -type x                            # ошибка: неверный тип
find -foo                               # ошибка: неизвестная опция

# ===== cal: все режимы =====
cal                 # текущий месяц
cal 2024            # весь 2024 год
cal 2 2024          # февраль 2024
cal -m 2 2024       # неделя начинается с понедельника
cal 13 2024         # ошибка: месяц вне диапазона
cal abc             # ошибка: не число
cal 1 2 3           # ошибка: слишком много аргументов

# ===== прочее =====
unknown_command     # ошибка: неизвестная команда
exit
